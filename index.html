function kirimKeGoogleSheets() { // Ambil data yang sama seperti saat menyimpan ke localStorage const dataAplikasi = { identitas: { kelas: document.getElementById('namaKelas').value, mapel: document.getElementById('namaMapel').value, semester: document.getElementById('semesterInfo').value, sekolah: document.getElementById('namaSekolah').value, guru: document.getElementById('namaGuru').value }, siswaList: [] }; document.querySelectorAll('#tableInput tbody tr').forEach(tr => { dataAplikasi.siswaList.push({ nama: tr.querySelector('.nama-siswa').value, jwbPG: tr.querySelector('.jwb-pg').value.split(',').map(s => s.trim()), skorSingkat: tr.querySelector('.skor-singkat').value, skorEssay: tr.querySelector('.skor-essay').value }); }); // URL Web App dari Google Apps Script Anda const urlAppsScript = "https://script.google.com/macros/s/AKfycby7gwy9oz7wfw5iTXCCwmCMrSc5la-u2ABfYJ-8BptT_Y-9Z5g3tbh50Xq5RdC27g13gw/exec"; fetch(urlAppsScript, { method: "POST", mode: "no-cors", // Diperlukan untuk lintas domain Google Scripts headers: { "Content-Type": "application/json" }, body: JSON.stringify(dataAplikasi) }) .then(() => { alert("✓ Data berhasil dikirim ke Google Sheets!"); }) .catch(error => { console.error("Error:", error); alert("❌ Gagal mengirim data ke Google Sheets."); }); }
